package io.github.khopland.versionchecker

import com.intellij.openapi.application.ApplicationManager
import com.intellij.openapi.command.WriteCommandAction
import com.intellij.openapi.fileEditor.FileDocumentManager
import com.intellij.testFramework.PlatformTestUtil
import com.intellij.testFramework.fixtures.BasePlatformTestCase
import io.github.khopland.versionchecker.core.*
import io.github.khopland.versionchecker.gradle.*
import io.github.khopland.versionchecker.npm.*
import kotlinx.coroutines.runBlocking
import org.jetbrains.plugins.gradle.settings.GradleProjectSettings
import org.jetbrains.plugins.gradle.settings.GradleSettings
import java.util.concurrent.Callable

/** Observation probes: assertions describe present defects, not desired behavior. */
class ProjectReviewProbeTest : BasePlatformTestCase() {
    private fun <T> background(action: suspend () -> T): T = PlatformTestUtil.waitForFuture(
        ApplicationManager.getApplication().executeOnPooledThread(Callable { runBlocking { action() } }), 30_000)

    override fun tearDown() {
        try {
            val settings = GradleSettings.getInstance(project)
            settings.linkedProjectsSettings.mapNotNull { it.externalProjectPath }.forEach(settings::unlinkExternalProject)
        } finally { super.tearDown() }
    }

    fun testNpmUpdateLeavesDirectOverrideIncompatible() {
        val file = myFixture.addFileToProject("override-case/package.json",
            """{"dependencies":{"alpha":"1.2.3"},"overrides":{"alpha":"1.2.3"}}""")
        val adapter = NpmBuildSystemAdapter()
        val snapshot = adapter.snapshot(project, file.virtualFile)!!
        val report = UpdateReport(listOf(UpdateCandidate(snapshot.declarations.single(), "1.2.9")))
        val plan = background { adapter.prepareUpdates(project, mapOf(snapshot to report)) }
        assertEquals(1, plan.changes.size)
        assertTrue(plan.apply(project))
        assertTrue(file.text.contains("\"dependencies\":{\"alpha\":\"1.2.9\"}"))
        assertTrue(file.text.contains("\"overrides\":{\"alpha\":\"1.2.3\"}"))
    }

    fun testGradleBuildLogicEditsRemainCurrentAndApply() {
        val file = myFixture.addFileToProject("build-logic-case/build.gradle",
            "dependencies { implementation 'g:library:1.2.3' }")
        val logic = myFixture.addFileToProject("build-logic-case/buildSrc/src/main/kotlin/RepositoryPolicy.kt",
            "val approvedVersion = \"1.2.9\"")
        GradleSettings.getInstance(project).linkProject(GradleProjectSettings().apply {
            externalProjectPath = file.virtualFile.parent.path
        })
        FileDocumentManager.getInstance().saveAllDocuments()
        val adapter = GradleBuildSystemAdapter()
        val snapshot = background { adapter.discover(project, BuildSelection(UpdateScope.CURRENT_FILE, file.virtualFile.path)) }.single()
        assertFalse(snapshot.fingerprint.files.containsKey(logic.virtualFile.path))
        val report = UpdateReport(listOf(UpdateCandidate(snapshot.declarations.single(), "1.2.9")))
        val plan = background { adapter.prepareUpdates(project, mapOf(snapshot to report)) }
        WriteCommandAction.runWriteCommandAction(project) {
            FileDocumentManager.getInstance().getDocument(logic.virtualFile)!!.setText("val approvedVersion = \"1.2.3\"")
        }
        FileDocumentManager.getInstance().saveAllDocuments()
        assertTrue(background { adapter.isCurrentInBackground(project, snapshot) })
        assertTrue(plan.apply(project))
    }

    fun testNpmProjectInsideDirectoryNamedBuildIsExcluded() {
        val file = myFixture.addFileToProject("build/independent-project/package.json", "{}")
        assertFalse(NpmManifest.supported(file.virtualFile))
    }

    fun testGradleHugeMajorIsMisclassifiedAsPatch() {
        assertTrue(GradleVersions.newer("2147483648.1.0", "2147483649.1.0"))
        assertEquals(VersionChangeKind.PATCH, GradleVersions.kind("2147483648.1.0", "2147483649.1.0"))
    }

    fun testMalformedTypedCatalogThrowsInsteadOfSkipping() {
        val failure = runCatching { GradleDeclarations.parse("gradle/libs.versions.toml",
            "[libraries]\nalpha = { module = 123, version = \"1.2.3\" }\nbeta = \"g:b:1.2.3\"") }.exceptionOrNull()
        assertNotNull(failure)
        assertTrue(failure!!.javaClass.simpleName.contains("InvalidType"))
    }

    fun testGradleClassifierNotationVanishesFromCoverage() {
        val parsed = GradleDeclarations.parse("build.gradle", "dependencies { implementation 'g:a:1.2.3:tests' }")
        assertTrue(parsed.isEmpty())
    }
}
